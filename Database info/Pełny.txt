//////////////////////////////////////////////////////
// USERS
//////////////////////////////////////////////////////

Table users {
  id integer [pk, increment]
  user_type integer // 0 - admin, 1 - librarian, 2 - user
  first_name varchar
  last_name varchar
  email varchar [unique]
  phone varchar
  created_at datetime
}

//////////////////////////////////////////////////////
// AUTHORS
//////////////////////////////////////////////////////

Table authors {
  id integer [pk, increment]
  first_name varchar
  last_name varchar
  bio text
}

Table book_authors {
  id integer [pk, increment]
  book_id integer [ref: > books.id]
  author_id integer [ref: > authors.id]
}

//////////////////////////////////////////////////////
// BOOKS - general information about a book
//////////////////////////////////////////////////////

Table books {
  id integer [pk, increment]
  title varchar
  description text
  isbn varchar [unique]
  published_year integer
  category_id integer [ref: > categories.id]
}

//////////////////////////////////////////////////////
// CATEGORIES
//////////////////////////////////////////////////////

Table categories {
  id integer [pk, increment]
  name varchar
}

//////////////////////////////////////////////////////
// COPIES (physical items)
//////////////////////////////////////////////////////

Table books_physical {
  id integer [pk, increment]
  book_id integer [ref: > books.id]
  inventory_code varchar [unique]
  status varchar // available, borrowed, damaged, lost
}

//////////////////////////////////////////////////////
// LOANS (borrowed books)
//////////////////////////////////////////////////////

Table current_loans_status {
  id integer [pk, increment]
  user_id integer [ref: > users.id]
  copy_id integer [ref: > books_physical.id]
  loan_date datetime
  due_date datetime
  return_date datetime
  status varchar//active, returned, overdue
}

//////////////////////////////////////////////////////
// RESERVATIONS - reservations/books in users "shoping cart"
//////////////////////////////////////////////////////

Table reservations {
  id integer [pk, increment]
  user_id integer [ref: > users.id]
  copy_id integer [ref: > books_physical.id]
  reserved_at datetime
  status varchar // active, fulfilled, cancelled
}

//////////////////////////////////////////////////////
// FEES
//////////////////////////////////////////////////////

Table fees {
  id integer [pk, increment]
  user_id integer [ref: > users.id]
  loan_id integer [ref: > current_loans_status.id]
  amount decimal(18,2)
  created_at datetime
  paid_at datetime
  status varchar
}




///////////////////////////////////////////////////////////////////////////
// dodatkowo
//////

//////////////////////////////////////////////////////
// COPY STATUS HISTORY
//////////////////////////////////////////////////////

Table copy_status_history {
  id integer [pk, increment]
  copy_id integer [ref: > books_physical.id]
  old_status varchar
  new_status varchar
  changed_at_date datetime
  changed_by_user integer [ref: > users.id] // kto zmienił status
  note text
}




//////////////////////////////////////////////////////
// LOAN EVENTS
/////////////////////////////////////////////////////

Table loan_events {
  id integer [pk, increment]
  loan_id integer [ref: > current_loans_status.id]
  event_type varchar// borrowed, extended, returned, overdue, fee_assigned
  event_date datetime
  user_id integer [ref: > users.id]
  changed_by_user integer [ref: > users.id] // kto zmienił status, może być że System(automatycznie)
  note text
}


//////////////////////////////////////////////////////
// ADMIN_LOGS
//////////////////////////////////////////////////////

//W bibliotece ważne jest:
//-kto dodał książkę,
//-kto usunął egzemplarz,
//-kto zmienił status,
//-kto zmienił dane użytkownika.Dep

Table admin_logs{
  id integer [pk, increment]
  user_id integer [ref: > users.id]
  action varchar //add_book, update_book, delete_book, add_copy, update_copy_status, delete_copy, update_user, block_user, unblock_user, create_fee, override_loan, system_auto_overdue
  entity_type text // typ obiektu, na którym wykonano operację: book, copy, user, loan, reservation, fee, category, tag
  entity_id integer // ID obiektu, którego dotyczy operacja. Jest zwykłym integerem, nie ma relacji, bo logi muszą działać nawet po usunięciu obiektu.
  created_at datetime

}


//////////////////////////////////////////////////////
// REVIEWS
//////////////////////////////////////////////////////
//reviews(id, user_id, book_id, rating, comment, created_at)

Table reviews{
  id integer [pk, increment]
  user_id integer [ref: > users.id]
  book_id integer [ref: > books.id]
  rating integer
  comment text
  created_at datetime
}



//////////////////////////////////////////////////////

//tags(id, name)
Table tags{
  id integer [pk, increment]
  name text
}

//////////////////////////////////////////////////////
// BOOK_TAGS
//////////////////////////////////////////////////////
//book_tags(book_id, tag_id)
Table book_tags{
  book_id integer [ref: > books.id]

}
