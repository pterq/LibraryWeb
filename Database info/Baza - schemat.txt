//////////////////////////////////////////////////////
// USERS - information about users
//////////////////////////////////////////////////////

Table users {
  id integer [pk, increment]
  role varchar [not null] // admin, librarian, user
  first_name varchar
  last_name varchar
  email varchar [unique]
  phone varchar
  password_hash varchar [not null] //hashed password 
  created_at datetime
}


//////////////////////////////////////////////////////
// AUTHORS - author information
//////////////////////////////////////////////////////

Table authors {
  id integer [pk, increment]
  first_name varchar [not null]
  last_name varchar [not null]
  bio text
}

Table book_authors {
  id integer [pk, increment]
  book_id integer [ref: > books.id]
  author_id integer [ref: > authors.id]
}

//////////////////////////////////////////////////////
// BOOKS - general information about a book
//////////////////////////////////////////////////////

Table books {
  id integer [pk, increment]
  title varchar [not null]
  description text
  isbn varchar [unique]
  published_year integer
  category_id integer [ref: > categories.id]
}

//////////////////////////////////////////////////////
// CATEGORIES - names of book genres
//////////////////////////////////////////////////////

Table categories {
  id integer [pk, increment]
  name varchar
}

//////////////////////////////////////////////////////
// COPIES -  physical items/books
//////////////////////////////////////////////////////

Table books_physical {
  id integer [pk, increment]
  book_id integer [ref: > books.id]
  inventory_code varchar [unique]
  status varchar [not null] // available, reserved, borrowed, damaged, lost
}

//////////////////////////////////////////////////////
// LOANS - borrowed books history and status
//////////////////////////////////////////////////////

Table current_loans_status {
  id integer [pk, increment]
  user_id integer [ref: > users.id]
  copy_id integer [ref: > books_physical.id]
  loan_date datetime
  due_date datetime
  return_date datetime
  status varchar [not null] //active, returned, overdue
}

//////////////////////////////////////////////////////
// RESERVATIONS - reservations/books in user's "shoping cart"
//////////////////////////////////////////////////////

Table reservations {
  id integer [pk, increment]
  user_id integer [ref: > users.id]
  copy_id integer [ref: > books_physical.id]
  reserved_at datetime [not null]
  expires_at datetime [not null]
  status varchar [not null] // active, fulfilled, cancelled
}

//////////////////////////////////////////////////////
// FEES - fee information
//////////////////////////////////////////////////////

Table fees {
  id integer [pk, increment]
  user_id integer [ref: > users.id]
  loan_id integer [ref: > current_loans_status.id]
  amount decimal(18,2) [not null]
  created_at datetime [not null]
  paid_at datetime
  status varchar [not null] // pending, paid, cancelled
}
